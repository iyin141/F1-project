# SECTION 4 — RATE LIMITING

```
🔴 Listed as item 2 in the header but never implemented

Install:
  pip install django-ratelimit

Two layers of rate limiting:

LAYER 1 — Per-IP rate limit on all API endpoints
LAYER 2 — Per-API-key rate limit for authenticated clients

Layer 1 — create api/middleware/rate_limit.py:

from django_ratelimit.decorators import ratelimit
from django.http import JsonResponse
from functools import wraps

def api_rate_limit(view_func):
    """
    60 requests per minute per IP for anonymous access.
    Applied to all /api/* endpoints.
    """
    @ratelimit(key='ip', rate='60/m', block=False)
    @wraps(view_func)
    def wrapped(request, *args, **kwargs):
        if getattr(request, 'limited', False):
            return JsonResponse({
                'error': 'Rate limit exceeded',
                'error_code': 'RATE_LIMITED',
                'retry_after_seconds': 60,
            }, status=429)
        return view_func(request, *args, **kwargs)
    return wrapped


Layer 2 — Per-API-key rate limit (higher limit for known clients):

def api_key_rate_limit(view_func):
    """
    300 requests per minute per API key.
    Only applied when X-Api-Key header is present and valid.
    """
    @ratelimit(key='header:x-api-key', rate='300/m', block=False)
    @wraps(view_func)
    def wrapped(request, *args, **kwargs):
        if getattr(request, 'limited', False):
            return JsonResponse({
                'error': 'API key rate limit exceeded',
                'error_code': 'RATE_LIMITED',
                'retry_after_seconds': 60,
            }, status=429)
        return view_func(request, *args, **kwargs)
    return wrapped


Apply to views in api/views.py:

from api.middleware.rate_limit import api_rate_limit

class SeasonScheduleView(APIView):
    @api_rate_limit
    def get(self, request, year):
        ...


Add RATELIMIT settings to settings.py:
  RATELIMIT_ENABLE  = config('RATE_LIMIT_ENABLED', default=True, cast=bool)
  RATELIMIT_USE_CACHE = 'default'
  CACHES = {
      'default': {
          'BACKEND': 'django.core.cache.backends.locmem.LocMemCache',
      }
  }
```

# SECTION 5 — API KEY VERIFICATION

```
🔴 Listed as item 3 in the header but never implemented

Strategy: simple header-based API key for the Next.js
frontend to authenticate with the Django backend.
Prevents the Django API from being called directly by
third parties in production.

Step 1 — create api/middleware/api_key.py:

from django.http import JsonResponse
from decouple import config

INTERNAL_API_KEY = config('INTERNAL_API_KEY', default='')

class ApiKeyMiddleware:
    """
    Validates X-Api-Key header on all /api/* requests.
    Passes through OPTIONS requests (CORS preflight).
    In DEBUG mode, bypasses validation for local development.
    """

    def __init__(self, get_response):
        self.get_response = get_response

    def __call__(self, request):
        # Allow CORS preflight through
        if request.method == 'OPTIONS':
            return self.get_response(request)

        # Skip auth in debug mode for local dev
        if config('DEBUG', default=False, cast=bool):
            return self.get_response(request)

        # Only validate /api/* routes
        if not request.path.startswith('/api/'):
            return self.get_response(request)

        provided_key = request.headers.get('X-Api-Key', '')
        if not provided_key or provided_key != INTERNAL_API_KEY:
            return JsonResponse({
                'error': 'Invalid or missing API key',
                'error_code': 'UNAUTHORIZED',
            }, status=401)

        return self.get_response(request)


Step 2 — add to MIDDLEWARE in settings.py:
  MIDDLEWARE = [
      'corsheaders.middleware.CorsMiddleware',
      'api.middleware.api_key.ApiKeyMiddleware',  # add here
      'django.middleware.common.CommonMiddleware',
      ...
  ]


Step 3 — Next.js route handlers send the key:

// app/api/races/[year]/route.ts

# SECTION 6 — REQUEST TIMEOUT HANDLING

```
🔴 Not implemented — FastF1 calls can take 30-120 seconds
   on a cold load. Without timeouts, requests hang forever.

Wrap all FastF1 calls with a timeout:

# api/services/fastf1_runtime.py — add timeout wrapper

import signal
from contextlib import contextmanager

class FastF1TimeoutError(Exception):
    pass

@contextmanager
def fastf1_timeout(seconds: int = 90):
    """
    Raises FastF1TimeoutError if FastF1 call exceeds timeout.
    Uses SIGALRM — Unix only (works on Railway, Render, fly.io)
    """
    def handler(signum, frame):
        raise FastF1TimeoutError(
            f"FastF1 call exceeded {seconds}s timeout"
        )
    signal.signal(signal.SIGALRM, handler)
    signal.alarm(seconds)
    try:
        yield
    finally:
        signal.alarm(0)


Usage in services:

def _fetch_results(year, round_num):
    with fastf1_timeout(seconds=90):
        session = fastf1.get_session(year, round_num, 'Race')
        session.load(...)
        ...


Handle in views:

from api.services.fastf1_runtime import FastF1TimeoutError

try:
    data = get_race_results(year, round_num)
except FastF1TimeoutError:
    return Response({
        'error': 'Data source timeout — try again shortly',
        'error_code': 'UPSTREAM_TIMEOUT',
        'readiness': {
            'can_proceed': False,
            'message': 'FastF1 request timed out',
        }
    }, status=504)

# SECTION 7 — CACHE INVALIDATION ENDPOINT

```
🔴 Not implemented — no way to clear stale cache entries
   without direct DB access

Add admin-only endpoint for cache management:

# api/views/cache_views.py

from rest_framework.views import APIView
from rest_framework.response import Response
from api.services import db_cache
from decouple import config

ADMIN_KEY = config('ADMIN_API_KEY', default='')

class CacheInvalidateView(APIView):
    """
    POST /api/admin/cache/invalidate/
    Body: { "key": "schedule:2024" }
          or { "season": 2024 }
          or { "all": true }

    Protected by admin API key — different from the
    internal frontend key.
    """

    def post(self, request):
        admin_key = request.headers.get('X-Admin-Key', '')
        if admin_key != ADMIN_KEY:
            return Response({'error': 'Unauthorized'}, status=401)

        body = request.data

        if body.get('all'):
            from api.models import DataCache
            count = DataCache.objects.all().delete()[0]
            return Response({'deleted': count, 'scope': 'all'})

        if 'season' in body:
            count = db_cache.invalidate_season(int(body['season']))
            return Response({'deleted': count, 'scope': f"season:{body['season']}"})

        if 'key' in body:
            deleted = db_cache.invalidate(body['key'])
            return Response({'deleted': 1 if deleted else 0, 'key': body['key']})

        return Response({'error': 'Provide key, season, or all'}, status=400)


class CacheStatsView(APIView):
    """
    GET /api/admin/cache/stats/
    Returns count, oldest entry, newest entry, stale count
    """

    def get(self, request):
        admin_key = request.headers.get('X-Admin-Key', '')
        if admin_key != ADMIN_KEY:
            return Response({'error': 'Unauthorized'}, status=401)

        from api.models import DataCache
        from django.utils import timezone

        entries = DataCache.objects.all()
        stale   = [e for e in entries if e.is_stale]

        return Response({
            'total_entries':  entries.count(),
            'stale_entries':  len(stale),
            'oldest_entry':   entries.order_by('refreshed_at').first().cache_key
                              if entries.exists() else None,
            'newest_entry':   entries.order_by('-refreshed_at').first().cache_key
                              if entries.exists() else None,
        })


Register in urls.py:
  path('api/admin/cache/invalidate/', CacheInvalidateView.as_view()),
  path('api/admin/cache/stats/',      CacheStatsView.as_view()),
```

# api/views/health_views.py

from rest_framework.views import APIView
from rest_framework.response import Response
from django.db import connection

class HealthView(APIView):
    """
    GET /api/health/
    Returns 200 if service is healthy.
    Returns 503 if DB is unreachable.
    No authentication required — public endpoint.
    """
    authentication_classes = []
    permission_classes      = []

    def get(self, request):
        checks = {}

        # DB check
        try:
            with connection.cursor() as cursor:
                cursor.execute("SELECT 1")
            checks['database'] = 'ok'
        except Exception as e:
            checks['database'] = f'error: {str(e)}'

        # Cache check
        try:
            from api.models import DataCache
            checks['cache_entries'] = DataCache.objects.count()
            checks['cache'] = 'ok'
        except Exception as e:
            checks['cache'] = f'error: {str(e)}'

        all_ok = all(v == 'ok' or isinstance(v, int)
                     for v in checks.values())

        return Response(
            {'status': 'healthy' if all_ok else 'degraded', **checks},
            status=200 if all_ok else 503
        )
