
`api/middleware/rate_limit.py`:
```python
from django_ratelimit.decorators import ratelimit
from django.http import JsonResponse
from functools import wraps

def api_rate_limit(view_func):
    @ratelimit(key='ip', rate='60/m', block=False)
    @wraps(view_func)
    def wrapped(request, *args, **kwargs):
        if getattr(request, 'limited', False):
            return JsonResponse({'error': 'Rate limit exceeded', 'retry_after_seconds': 60}, status=429)
        return view_func(request, *args, **kwargs)
    return wrapped

def api_key_rate_limit(view_func):
    @ratelimit(key='header:x-api-key', rate='300/m', block=False)
    @wraps(view_func)
    def wrapped(request, *args, **kwargs):
        if getattr(request, 'limited', False):
            return JsonResponse({'error': 'API key rate limit exceeded', 'retry_after_seconds': 60}, status=429)
        return view_func(request, *args, **kwargs)
    return wrapped
```

## §5 — API KEY VERIFICATION `🔴`

`api/middleware/api_key.py`:
```python
from django.http import JsonResponse
from decouple import config

INTERNAL_API_KEY = config('INTERNAL_API_KEY', default='')

class ApiKeyMiddleware:
    def __init__(self, get_response):
        self.get_response = get_response

    def __call__(self, request):
        if request.method == 'OPTIONS': return self.get_response(request)
        if request.path == '/api/health/': return self.get_response(request)
        if config('DEBUG', default=False, cast=bool): return self.get_response(request)
        if not request.path.startswith('/api/'): return self.get_response(request)

        if request.headers.get('X-Api-Key', '') != INTERNAL_API_KEY:
            return JsonResponse({'error': 'Invalid or missing API key'}, status=401)
        return self.get_response(request)
```

Add to `MIDDLEWARE` after `CorsMiddleware`. Next.js sends key via `X-Api-Key` header from server-side env var `DJANGO_API_KEY` (no `NEXT_PUBLIC_` prefix).

## §6 — REQUEST TIMEOUT `🔴`

`api/services/fastf1_runtime.py`:
```python
import signal
from contextlib import contextmanager

class FastF1TimeoutError(Exception): pass

@contextmanager
def fastf1_timeout(seconds=90):
    def handler(signum, frame): raise FastF1TimeoutError()
    signal.signal(signal.SIGALRM, handler)
    signal.alarm(seconds)
    try: yield
    finally: signal.alarm(0)
```

Views catch `FastF1TimeoutError` → return `504`.

## §19 — NO 500s POLICY `🔴`

Every endpoint must return a structured JSON error — never an unhandled exception.

**Wrap all views:**

```python
# api/utils/responses.py

from rest_framework.response import Response

def error_response(message, code, status, extra=None):
    body = {'error': message, 'error_code': code}
    if extra:
        body.update(extra)
    return Response(body, status=status)
```

**Global exception handler** — catches anything that slips through:

`api/utils/exception_handler.py`:
```python
from rest_framework.views import exception_handler
from rest_framework.response import Response
import logging

logger = logging.getLogger('api')

def custom_exception_handler(exc, context):
    response = exception_handler(exc, context)

    if response is not None:
        return response

    # Unhandled exception — log it, return 500 with no traceback exposed
    logger.exception(f"Unhandled exception in {context['view'].__class__.__name__}: {exc}")
    return Response({
        'error': 'An unexpected error occurred',
        'error_code': 'INTERNAL_ERROR',
    }, status=500)
```

Register in `settings.py`:
```python
REST_FRAMEWORK = {
    'EXCEPTION_HANDLER': 'api.utils.exception_handler.custom_exception_handler',
}
```

**Per-view pattern — every view follows this:**

```python
class RaceResultsView(APIView):
    @api_rate_limit
    def get(self, request, year, round_num):
        try:
            data = get_race_results(year, round_num)
        except FastF1TimeoutError:
            return error_response('Data source timeout', 'UPSTREAM_TIMEOUT', 504)
        except Exception as e:
            logger.exception(f"RaceResultsView failed: {e}")
            return error_response('Failed to fetch race results', 'FETCH_ERROR', 503)

        if data is None or not data:
            return error_response('No data available for this race', 'NO_DATA', 404)

        return Response(data)
```

**Rules:**

- `404` — resource exists in schema but no data found (e.g. race not yet run)
- `503` — data source (FastF1 / Neon) failed
- `504` — FastF1 timeout
- `429` — rate limited
- `401` — bad/missing API key
- `500` — only if global handler catches something truly unexpected, never thrown intentionally

**Never let these reach the view unhandled:**
- `fastf1` exceptions — `SessionNotAvailableError`, `DataNotLoadedError`, any `Exception` from `.load()`
- DB exceptions — `OperationalError`, `ProgrammingError`, `DataCache.DoesNotExist`
- `KeyError` / `IndexError` / `AttributeError` from parsing FastF1 response data
- `requests.Timeout`, `requests.ConnectionError` from any outbound HTTP calls

```python
# Defensive pattern for FastF1 data parsing
try:
    results = session.results[['DriverNumber', 'FullName', 'TeamName', 'Position']].to_dict('records')
except (KeyError, AttributeError, ValueError) as e:
    logger.warning(f"Failed to parse session results: {e}")
    return error_response('Data parsing failed', 'PARSE_ERROR', 503)
```
